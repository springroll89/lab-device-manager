# Slice A.2 — Multi-device App (timestamps + config cutover + dashboard + backfill) Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the phase-1 single-device prototype with a real multi-device web app backed by the A.1 SQLite layer: precise (epoch-ms, single-clock-aligned) timestamps, a multi-card live dashboard, a recorded-runs list, and a backfill (tagging) UI.

**Architecture:** The app loads a multi-device config, runs one `Engine` (A.1) writing runs/samples/events to a `Repository` (A.1), and a Flask server serves a multi-card dashboard (live SSE per device) + a runs list + backfill form. **All timestamps become epoch-milliseconds (UTC)** taken at read time on the single collector → cross-device aligned. Phase-1's single-device config / `web/app.py` / `__main__.py` / `session_logger.py` (CSV) are retired and replaced.

**Tech Stack:** Python 3.14, stdlib `sqlite3`/`tomllib`, Flask, pytest. Reuses A.1 (`pump_monitor/db`, `pump_monitor/runtime`) + phase-1 采集层 (`instruments`, `sampler`, `modbus_io`).

## Global Constraints

- **Precise timestamps:** every run(sample/event timestamp is **epoch milliseconds (UTC int)**, captured as `int(time.time()*1000)` at read. Display converts to ISO (UTC + local). Cross-device alignment is inherent (one collector reads all → one clock). Document the NTP assumption if multiple collectors ever exist.
- **READ-ONLY** (no device writes). Reuse A.1 db/runtime + phase-1 采集层 unchanged in behavior.
- **SQLite** (`Repository`, persistent single connection, `check_same_thread=False` + `busy_timeout`).
- Multi-device via `[[devices]]` in config; validated with the 1 live pump.
- **Cutover:** A.2 retires phase-1's `config.py`/`web/app.py`/`__main__.py`/`session_logger.py` and their tests. At every commit the suite stays green; the new app is runnable after Task 5.
- Commits per task; `<type>: <desc>`; **no Co-Authored-By**. `.venv/bin/pytest`, ≥80%.
- XSS-safe frontend (`textContent`, no `innerHTML` with data). Frozen dataclasses.
- Branch `feat/slice-a2-app` off `main` (which now has phase-1 + A.1).

## Scope (explicit — adjust before executing)
- **In:** epoch-ms timestamps; multi-device config; Engine+Repository-backed app; multi-card live dashboard; recorded-runs list; backfill (operator/project/experiment/remark) UI.
- **Out (later slices):** run-detail event timeline + flow curve (B); calibration ledger (C); maintenance (C); Feishu alerting + stats (D); phase-2 write control (E).

def fmt_ts_ms(ms: int) -> str:
    """Epoch ms -> 'YYYY-MM-DDTHH:MM:SS (local) / UTC' for display."""
    import time as _t
    s = ms / 1000.0
    return _t.strftime("%Y-%m-%d %H:%M:%S", _t.localtime(s))

    def open_run(self, device_id, channel, started_ms, setpoints):
        c = self._conn.execute(
            """INSERT INTO run(device_id, channel, started_ms,
                  setpoints_json, work_mode, target_volume, target_volume_unit)
               VALUES(?,?,?,?,?,?,?)""",

             setpoints.get("work_mode"), setpoints.get("target_volume"),

    def close_run(self, run_id, ended_ms, end_status, result_acc_volume, result_acc_unit, alarm_count):

from __future__ import annotations
import tomllib
from dataclasses import dataclass
from pump_monitor.runtime.types import DeviceConfig


@dataclass(frozen=True)
class Config:
    db_path: str = "./data/pump_monitor.db"
    sample_interval_ms: int = 1000
    web_port: int = 7800
    auto_open_browser: bool = True
    devices: tuple = ()   # tuple[DeviceConfig]


def load_config(path: str = "config.toml") -> Config:
    with open(path, "rb") as f:
        d = tomllib.load(f)
    devs = tuple(DeviceConfig(
        name=dev["name"], type=dev["type"], alias=dev.get("alias", ""),
        serial_port=dev.get("serial_port", ""), baudrate=dev.get("baudrate", 9600),
        parity=dev.get("parity", "EVEN"), modbus_addr=dev.get("modbus_addr", 1),
        wordorder=dev.get("wordorder", "CDAB"), channel=dev.get("channel", 1),
    ) for dev in d.get("devices", []))
    return Config(
        db_path=d.get("db_path", "./data/pump_monitor.db"),
        sample_interval_ms=d.get("sample_interval_ms", 1000),
        web_port=d.get("web_port", 7800),
        auto_open_browser=d.get("auto_open_browser", True),
        devices=devs,
    )

`pump_monitor/web/app.py` (full rewrite):
```python
from __future__ import annotations
from flask import Flask, jsonify, request, send_from_directory
from pump_monitor.db.models import fmt_ts_ms


def _snap_to_dict(snap):
    return {
        "state": snap.state, "work_mode": snap.work_mode, "flow_rpm": snap.flow_rpm,
        "acc_volume": snap.acc_volume, "acc_unit": snap.acc_unit, "temp_c": snap.temp_c,
        "progress_pct": snap.progress_pct, "remaining_ms": snap.remaining_ms,
        "alarm": snap.alarm, "ts_ms": int(snap.timestamp * 1000),
    }


def _run_to_dict(run):
    return {
        "id": run.id, "device_id": run.device_id,
        "started_ms": run.started_ms, "ended_ms": run.ended_ms,
        "duration_ms": run.duration_ms, "end_status": run.end_status,
        "operator": run.operator, "project_tag": run.project_tag,
        "tagged": run.tagged, "alarm_count": run.alarm_count,
        "result_acc_volume": run.result_acc_volume, "result_acc_unit": run.result_acc_unit,
        "started_display": fmt_ts_ms(run.started_ms) if run.started_ms else "",
    }


def create_app(engine, repo):
    app = Flask(__name__, static_folder="static", static_url_path="/static")

    @app.get("/")
    def index():
        return send_from_directory(app.static_folder, "index.html")

    @app.get("/api/status")
    def api_status():
        latest = engine.latest()
        dmap = engine.device_map()
        devices = []
        for did, snap in latest.items():
            dc = dmap.get(did)
            devices.append({"id": did, "name": dc.name if dc else str(did),
                            "alias": dc.alias if dc else "", "type": dc.type if dc else "",
                            "latest": _snap_to_dict(snap)})
        return jsonify({"devices": devices})

    @app.get("/api/runs")
    def api_runs():
        limit = int(request.args.get("limit", 50))
        rows = repo._conn.execute(
            "SELECT * FROM run ORDER BY started_ms DESC LIMIT ?", (limit,)).fetchall()
        from pump_monitor.db.repository import _row_to_run
        return jsonify([_run_to_dict(_row_to_run(r)) for r in rows])

    @app.get("/api/runs/untagged")
    def api_untagged():
        from pump_monitor.db.repository import _row_to_run
        rows = repo._conn.execute(
            "SELECT * FROM run WHERE tagged=0 AND ended_ms IS NOT NULL ORDER BY started_ms DESC").fetchall()
        return jsonify([_run_to_dict(_row_to_run(r)) for r in rows])

    @app.post("/api/runs/<int:run_id>/tag")
    def api_tag(run_id):
        body = request.get_json(silent=True) or {}
        repo.tag_run(run_id, body.get("operator", ""), body.get("project_tag", ""),
                     body.get("experiment_tag", ""), body.get("remark", ""))
        return jsonify({"ok": True})

    return app
```

`pump_monitor/__main__.py`:
```python
from __future__ import annotations
import threading
import webbrowser

from pump_monitor.config import load_config
from pump_monitor.db.repository import Repository
from pump_monitor.runtime.engine import Engine, serial_adapter_factory
from pump_monitor.web.app import create_app


def main():  # pragma: no cover - wiring + server lifecycle
    cfg = load_config()
    repo = Repository(cfg.db_path)
    eng = Engine(repo, list(cfg.devices), cfg.sample_interval_ms / 1000.0,
                 adapter_factory=serial_adapter_factory)
    eng.start()
    app = create_app(eng, repo)
    try:
        if cfg.auto_open_browser:
            url = f"http://127.0.0.1:{cfg.web_port}/"
            threading.Timer(1.0, lambda: webbrowser.open(url)).start()
        app.run(host="127.0.0.1", port=cfg.web_port, threaded=True, use_reloader=False)
    finally:
        eng.stop()
        repo.close()


if __name__ == "__main__":
    main()
```

`pump_monitor/web/static/app.js` (polls `/api/status` every 1s + `/api/runs` every 3s; 补录 via `prompt()` — XSS-safe, upgradeable to an inline form later):
```javascript
const $ = (id) => document.getElementById(id);
const fmtDur = (ms) => (ms == null) ? "-" : (ms >= 60000 ? (ms/60000).toFixed(1)+" 分" : (ms/1000).toFixed(0)+" 秒");

async function pollStatus() {
  let d;
  try { d = await (await fetch("/api/status")).json(); } catch (e) { return; }
  const box = $("devices"); box.textContent = "";
  for (const dev of (d.devices || [])) {
    const L = dev.latest || {};
    const card = document.createElement("div"); card.className = "card";
    const name = document.createElement("div"); name.className = "name"; name.textContent = dev.alias || dev.name;
    const st = document.createElement("div"); st.className = "v s-" + (L.state || "offline"); st.textContent = L.state || "offline";
    const flow = document.createElement("div"); flow.className = "v"; flow.textContent = (L.flow_rpm == null ? "-" : (+L.flow_rpm).toFixed(2)) + " rpm";
    const acc = document.createElement("div"); acc.className = "k"; acc.textContent = "累计 " + (L.acc_volume == null ? "-" : L.acc_volume) + " " + (L.acc_unit || "");
    const temp = document.createElement("div"); temp.className = "k"; temp.textContent = "温度 " + (L.temp_c == null ? "-" : L.temp_c) + " ℃";
    card.append(name, st, flow, acc, temp); box.appendChild(card);
  }
}

async function loadRuns() {
  let runs = [];
  try { runs = await (await fetch("/api/runs?limit=30")).json(); } catch (e) { return; }
  const tb = $("runs").querySelector("tbody"); tb.textContent = "";
  for (const r of runs) {
    const tr = document.createElement("tr");
    const cell = (txt, cls) => { const td = document.createElement("td"); if (cls) td.className = cls; td.textContent = (txt == null || txt === "") ? "-" : txt; return td; };
    tr.appendChild(cell(r.started_display));
    tr.appendChild(cell(r.device_id));
    tr.appendChild(cell(fmtDur(r.duration_ms)));
    tr.appendChild(cell(r.end_status, "s-" + (r.end_status === "completed" ? "running" : "stopped")));
    tr.appendChild(cell((r.result_acc_volume == null ? "-" : r.result_acc_volume) + " " + (r.result_acc_unit || "")));
    tr.appendChild(cell(r.operator));
    tr.appendChild(cell(r.project_tag));
    const act = document.createElement("td");
    if (!r.tagged) {
      const btn = document.createElement("button"); btn.textContent = "补录";
      btn.onclick = () => tagRun(r.id);
      act.appendChild(btn);
    } else { act.textContent = "✓"; }
    tr.appendChild(act);
    tb.appendChild(tr);
  }
}

async function tagRun(runId) {
  const operator = prompt("操作人？");
  if (operator === null) return;
  const project = prompt("项目 / 实验 tag？");
  if (project === null) return;
  await fetch("/api/runs/" + runId + "/tag", {
    method: "POST", headers: {"Content-Type": "application/json"},
    body: JSON.stringify({operator: operator || "", project_tag: project || "", experiment_tag: "", remark: ""})
  });
  loadRuns();
}

pollStatus(); loadRuns();
setInterval(pollStatus, 1000);
setInterval(loadRuns, 3000);
```
